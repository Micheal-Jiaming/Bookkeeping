# Versions are the ones actually installed and tested in .venv on 2026-08-23.
#
# The interface is Tkinter, which ships with Python -- there is no GUI dependency
# to install. Since 1.2.0 there is no web framework either.
anthropic==1.0.0
pillow==12.3.0

# Declared explicitly even though `anthropic` already depends on it. Our own
# `app/extract/base.py` imports BaseModel and Field directly, so this is a real
# first-order dependency of this project, not something inherited -- and relying
# on a transitive install means the app breaks the day anthropic changes its
# validation library. Pinned to the version in .venv on 2026-09-03.
pydantic==2.13.4

# Offline OCR, and the reason the app works with no API key and no install:
# these are the Python bindings for Windows' own OCR engine, which ships with
# Windows 10 and 11. They are bindings only -- nothing is downloaded at runtime
# and no model is bundled, because the recogniser is already part of the OS.
winrt-runtime==3.2.1
winrt-Windows.Foundation==3.2.1
winrt-Windows.Foundation.Collections==3.2.1
winrt-Windows.Globalization==3.2.1
winrt-Windows.Graphics.Imaging==3.2.1
winrt-Windows.Media.Ocr==3.2.1
winrt-Windows.Storage.Streams==3.2.1

# Optional second offline engine. pytesseract is only a wrapper -- it also needs
# the Tesseract binary installed (https://github.com/UB-Mannheim/tesseract/wiki).
# The application runs without it; the Tesseract engine just reports itself as
# unavailable and the Windows engine covers the offline case.
pytesseract==0.3.13

# Test-only
pytest==9.1.1

# Build-only: produces dist\Bookkeeping.exe from Bookkeeping.spec (build.bat
# installs this on demand, so a machine that only runs the app does not need it).
pyinstaller==6.22.2

# --- optional, and only in the full build -----------------------------------
#
# RapidOCR reads receipts markedly better than the Windows engine: on the six
# receipts confirmed by hand it matched 73 printed lines against 63, missed 4
# against 14, invented none against 3, and got 59 item names exactly right
# against 37. It is PaddleOCR's PP-OCR models converted to ONNX, so it needs
# neither PaddlePaddle nor PyTorch, runs on the CPU and never goes online.
#
# NOT installed by default, because it costs 95 MB in the frozen executable and
# about 1.7 seconds of start-up, and this application's first promise is one
# file you can copy onto a USB stick. `build.bat --full` installs these and
# bundles them; `build.bat` on its own does neither.
#
#   rapidocr==3.9.2
#   onnxruntime==1.29.0
